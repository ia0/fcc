Definition classjudg J :=
  match J with
    | JK k ⇒ cobj k CKind
    | JT t k ⇒ cobj t CType ∧ cobj k CKind
    | JP Y0 Y1 p ⇒ cobj Y0 CPEnv ∧ cobj Y1 CPEnv ∧ cobj p CProp
    | JC Y0 Y1 H' t' t ⇒ cobj Y0 CPEnv ∧ cobj Y1 CPEnv
                       ∧ cobj H' CTEnv ∧ cobj t' CType ∧ cobj t CType
    | JH H' ⇒ cobj H' CTEnv
    | JG G ⇒ cobj G CAEnv
    | Jwf o c ⇒ cobj o c
  end.
